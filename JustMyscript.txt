local function ScanForPan()

    if not ScriptRunning then
        return
    end

    local Character =
        LocalPlayer.Character

    if not Character then

        CurrentPan = nil
        CurrentCollect = nil
        CurrentShake = nil
        CurrentPanRemote = nil

        PanStatus.Text =
            "Pan: None equipped"

        return
    end

    local FoundPan = nil
    local FoundCollect = nil
    local FoundShake = nil
    local FoundPanRemote = nil

    for _, Object in ipairs(
        Character:GetChildren()
    ) do

        if Object:IsA("Tool") then

            local Scripts =
                Object:FindFirstChild("Scripts")

            if Scripts then

                local Collect =
                    Scripts:FindFirstChild("Collect")

                local Shake =
                    Scripts:FindFirstChild("Shake")

                local Pan =
                    Scripts:FindFirstChild("Pan")

                if Collect or Shake or Pan then

                    FoundPan = Object

                    if Collect
                        and Collect:IsA("RemoteFunction")
                    then
                        FoundCollect = Collect
                    end

                    if Shake
                        and Shake:IsA("RemoteEvent")
                    then
                        FoundShake = Shake
                    end

                    if Pan
                        and Pan:IsA("RemoteFunction")
                    then
                        FoundPanRemote = Pan
                    end

                    break
                end
            end
        end
    end

    CurrentPan = FoundPan
    CurrentCollect = FoundCollect
    CurrentShake = FoundShake
    CurrentPanRemote = FoundPanRemote

    if CurrentPan then

        PanStatus.Text =
            "Pan: " .. CurrentPan.Name

        PanStatus.TextColor3 =
            Color3.fromRGB(
                180,
                180,
                185
            )

    else

        PanStatus.Text =
            "Pan: None equipped"

        PanStatus.TextColor3 =
            Color3.fromRGB(
                150,
                100,
                100
            )

    end
end

local function GetSand()

    local Pan = CurrentPan

    if not Pan or not Pan.Parent then

        ScanForPan()

        Pan = CurrentPan

    end

    if not Pan then
        return 0, 0
    end

    local Fill =
        Pan:GetAttribute("Fill") or 0

    local Stats =
        LocalPlayer:FindFirstChild("Stats")

    if not Stats then
        return Fill, 0
    end

    local Capacity =
        Stats:GetAttribute("Capacity") or 0

    return Fill, Capacity
end

local function UpdateSandDisplay()

    if not ScriptRunning then

    local Fill, Capacity =
        GetSand()

    SandStatus.Text =
        "Sand: "
        .. tostring(math.floor(Fill))
        .. " / "
        .. tostring(math.floor(Capacity))

end

local function Collect()

    if not ScriptRunning then
        return
    end

    if not CurrentPan
        or not CurrentPan.Parent
    then

        ScanForPan()

    end

    if not CurrentCollect then

        warn(
            "[Collect] Collect RemoteFunction not found."
        )

        return
    end

    local Args = {
        1.00000000000000000,
        false
    }

    local Success, Error =
        pcall(function()

            CurrentCollect:InvokeServer(
                unpack(Args)
            )

        end)

    if not Success then

        warn(
            "[Collect] Error:",
            Error
        )

    end
end

--==================================================
-- WATER PAN
--==================================================

local function StartWaterPan()

    if not ScriptRunning then
        return false
    end

    ScanForPan()

    if not CurrentPan then

        warn(
            "[Pan] No equipped pan."
        )

        return false
    end

    local Scripts =
        CurrentPan:FindFirstChild("Scripts")

    if not Scripts then

        warn(
            "[Pan] Scripts folder missing."
        )

        return false
    end

    local PanRemote =
        Scripts:FindFirstChild("Pan")

    if not PanRemote then

        warn(
            "[Pan] Pan RemoteFunction missing."
        )

        return false
    end

    if not PanRemote:IsA("RemoteFunction") then

        warn(
            "[Pan] Pan is "
            .. PanRemote.ClassName
        )

        return false
    end

    CurrentPanRemote = PanRemote

    local Character =
        LocalPlayer.Character

    if not Character then
        return false
    end

    print(
        "[Pan] Calling Pan:InvokeServer()"
    )

    local Success, Result =
        pcall(function()

            return PanRemote:InvokeServer()

        end)

    if not Success then

        warn(
            "[Pan] Pan InvokeServer error:",
            Result
        )

        return false
    end

    print(
        "[Pan] Pan result:",
        tostring(Result)
    )

    if Result ~= true then

        warn(
            "[Pan] Pan activation rejected."
        )

        return false
    end

    local Timeout =
        os.clock() + 5

    while
        ScriptRunning
        and (AutoFarm or AutoShake)
        and os.clock() < Timeout
    do

        if Character:GetAttribute(
            "Panning"
        ) == true then

            print(
                "[Pan] Panning = true"
            )

            return true
        end

        task.wait(0.05)

    end

    warn(
        "[Pan] Server accepted Pan, "
        .. "but Panning never became true."
    )

    return false
end

--==================================================
-- SHAKE WHILE PANNING
--==================================================

local function ShakeWhilePanning()

    if not ScriptRunning then
        return false
    end

    local Character =
        LocalPlayer.Character

    if not Character then
        return false
    end

    ScanForPan()

    if not CurrentShake then

        warn(
            "[Shake] Shake RemoteEvent missing."
        )

        return false
    end

    local Stats =
        LocalPlayer:FindFirstChild("Stats")

    local ShakeSpeed = 2

    if Stats then

        ShakeSpeed =
            Stats:GetAttribute(
                "ShakeSpeed"
            ) or 1

    end

    ShakeSpeed =
        math.max(
            ShakeSpeed,
            0.1
        )

    local Cooldown =
        0.3 / ShakeSpeed

    print(
        "[Shake] ShakeSpeed:",
        ShakeSpeed,
        "Cooldown:",
        Cooldown
    )

    while
        ScriptRunning
        and (AutoFarm or AutoShake)
        and Character:GetAttribute(
            "Panning"
        ) == true
    do

        if not CurrentShake
            or not CurrentShake.Parent
        then

            ScanForPan()

            if not CurrentShake then
                break
            end

        end

        pcall(function()

            CurrentShake:FireServer()

        end)

        task.wait(Cooldown)

    end

    return true
end

local GetInventorySellPrice
local SellEvent

pcall(function()

    GetInventorySellPrice =
        ReplicatedStorage
            :WaitForChild("Remotes")
            :WaitForChild("Shop")
            :WaitForChild("GetInventorySellPrice")

    SellEvent =
        ReplicatedStorage
            :WaitForChild("Remotes")
            :WaitForChild("Shop")
            :WaitForChild("SellAll")

end)

local function SellAll()

    if not ScriptRunning then
        return
    end

    if GetInventorySellPrice
        and GetInventorySellPrice.Parent
    then

        pcall(function()

            if GetInventorySellPrice:IsA(
                "RemoteFunction"
            ) then

                GetInventorySellPrice:InvokeServer()

            elseif GetInventorySellPrice:IsA(
                "RemoteEvent"
            ) then

                GetInventorySellPrice:FireServer()

            end

        end)
    end

    task.wait(0.1)

    if SellEvent
        and SellEvent.Parent
    then

        pcall(function()

            if SellEvent:IsA(
                "RemoteEvent"
            ) then

                SellEvent:FireServer()

            elseif SellEvent:IsA(
                "RemoteFunction"
            ) then

                SellEvent:InvokeServer()

            end

        end)

    else

        warn(
            "[Sell] SellAll not found."
        )

    end
end

local function StartAutoSell()

    if AutoSellThread then
        return
    end

    AutoSellThread =
        task.spawn(function()

            while
                AutoSell
                and ScriptRunning
            do

                SellAll()

                task.wait(
                    AUTO_SELL_DELAY
                )

            end

            AutoSellThread = nil

        end)
end

local function GetGroundY(Position, Character)
    local RayParams = RaycastParams.new()
    RayParams.FilterType = Enum.RaycastFilterType.Exclude

local function TweenSegment(RootPart, TargetPosition, Character)
    if not ScriptRunning then
        return false
    end

    if not RootPart or not RootPart.Parent then
        return false
    end

    local StartCFrame = RootPart.CFrame
    local StartPosition = StartCFrame.Position

    local Distance =
        (StartPosition - TargetPosition).Magnitude

    if Distance <= 0.05 then
        RootPart.CFrame = CFrame.new(TargetPosition)
        return true
    end

    local Duration = math.max(
        Distance / TWEEN_SPEED,
        MIN_TWEEN_DURATION
    )

    local StartTime = os.clock()

    while ScriptRunning do
        if not RootPart.Parent then
            return false
        end

        local Alpha =
            math.clamp(
                (os.clock() - StartTime) / Duration,
                0,
                1
            )

        local NewPosition =
            StartPosition:Lerp(
                TargetPosition,
                Alpha
            )

        -- Directly update the root instead of relying
        -- on TweenService controlling the HumanoidRootPart.
        RootPart.CFrame =
            CFrame.new(NewPosition)

        if Alpha >= 1 then
            break
        end

        task.wait()
    end

    if not ScriptRunning then
        return false
    end

    RootPart.CFrame =
        CFrame.new(TargetPosition)

    return true
end

local function TweenTo(Position)

    if not ScriptRunning then
        return false
    end

    local Character =
        LocalPlayer.Character

    if not Character then
        warn("[Movement] Character not found.")
        return false
    end

    local RootPart =
        Character:FindFirstChild("HumanoidRootPart")

    if not RootPart then
        warn("[Movement] HumanoidRootPart not found.")
        return false
    end

    print(
        "[Movement] Starting:",
        RootPart.Position,
        "->",
        Position
    )

    local StartPosition =
        RootPart.Position

    local HorizontalStart =
        Vector3.new(
            StartPosition.X,
            0,
            StartPosition.Z
        )

    local HorizontalTarget =
        Vector3.new(
            Position.X,
            0,
            Position.Z
        )

    local HorizontalDistance =
        (
            HorizontalStart
            - HorizontalTarget
        ).Magnitude

    local SegmentCount =
        math.max(
            1,
            math.ceil(
                HorizontalDistance
                / TWEEN_SEGMENT_DISTANCE
            )
        )

    --==================================================
    -- DETERMINE TRAVEL HEIGHT
    --==================================================

    local TravelHeight =
        math.max(
            StartPosition.Y,
            GetSafeHeight(
                Vector3.new(
                    StartPosition.X,
                    Position.Y,
                    StartPosition.Z
                ),
                Character
            )
        )

    --==================================================
    -- MOVE UP
    --==================================================

    if StartPosition.Y < TravelHeight - 0.1 then

        print(
            "[Movement] Raising to:",
            TravelHeight
        )

        local RaisedPosition =
            Vector3.new(
                StartPosition.X,
                TravelHeight,
                StartPosition.Z
            )

        if not TweenSegment(
            RootPart,
            RaisedPosition,
            Character
        ) then
            warn("[Movement] Failed while raising.")
            return false
        end
    end

    --==================================================
    -- HORIZONTAL MOVEMENT
    --==================================================

    for Index = 1, SegmentCount do

        if not ScriptRunning then
            return false
        end

        if not RootPart.Parent then
            return false
        end

        local Alpha =
            Index / SegmentCount

        local X =
            StartPosition.X
            + (
                Position.X
                - StartPosition.X
            ) * Alpha

        local Z =
            StartPosition.Z
            + (
                Position.Z
                - StartPosition.Z
            ) * Alpha

        local SegmentPosition =
            Vector3.new(
                X,
                TravelHeight,
                Z
            )

        if not TweenSegment(
            RootPart,
            SegmentPosition,
            Character
        ) then
            warn(
                "[Movement] Failed at segment",
                Index,
                "/",
                SegmentCount
            )

            return false
        end
    end

    --==================================================
    -- FINAL POSITION
    --==================================================

    local FinalPosition =
        Vector3.new(
            Position.X,
            Position.Y,
            Position.Z
        )

    if not TweenSegment(
        RootPart,
        FinalPosition,
        Character
    ) then
        warn("[Movement] Failed final descent.")
        return false
    end

    print(
        "[Movement] Arrived:",
        RootPart.Position
    )

    return true
end

    return nil
end

local function GetSafeHeight(Position, Character)

    local GroundY =
        GetGroundY(
            Position,
            Character
        )

    if not GroundY then

        return Position.Y

    end

    local Humanoid =
        Character:FindFirstChildOfClass(
            "Humanoid"
        )

    local RootPart =
        Character:FindFirstChild(
            "HumanoidRootPart"
        )

    local Height =
        GROUND_CLEARANCE

    if Humanoid and RootPart then

        Height =
            math.max(
                GROUND_CLEARANCE,
                Humanoid.HipHeight
                    + RootPart.Size.Y / 2
            )

    end

    local MinimumY =
        GroundY + Height

    return math.max(
        Position.Y,
        MinimumY
    )
end

local function TweenSegment(
    RootPart,
    TargetPosition,
    Character
)

    if not ScriptRunning then
        return false
    end

    local StartPosition =
        RootPart.Position

    local Distance =
        (
            StartPosition
            - TargetPosition
        ).Magnitude

    if Distance <= 0.05 then
        return true
    end

    local Duration =
        math.max(
            Distance / TWEEN_SPEED,
            0.05
        )

    local Tween =
        TweenService:Create(
            RootPart,

            TweenInfo.new(
                Duration,
                Enum.EasingStyle.Linear
            ),

            {
                CFrame =
                    CFrame.new(
                        TargetPosition
                    )
            }
        )

    local Finished = false
    local Connection = nil

    Connection =
        Tween.Completed:Connect(
            function()

                Finished = true

                if Connection then
                    Connection:Disconnect()
                    Connection = nil
                end

            end
        )

    Tween:Play()

    while
        ScriptRunning
        and not Finished
    do

        -- Check the ground underneath
        -- the character while moving.

        local CurrentPosition =
            RootPart.Position

        local GroundY =
            GetGroundY(
                CurrentPosition,
                Character
            )

        if GroundY then

            local Humanoid =
                Character:FindFirstChildOfClass(
                    "Humanoid"
                )

            local MinimumHeight =
                GROUND_CLEARANCE

            if Humanoid then

                MinimumHeight =
                    math.max(
                        GROUND_CLEARANCE,
                        Humanoid.HipHeight
                            + RootPart.Size.Y / 2
                    )

            end

            local MinimumY =
                GroundY + MinimumHeight

            if CurrentPosition.Y < MinimumY then

                Tween:Cancel()

                RootPart.CFrame =
                    CFrame.new(
                        CurrentPosition.X,
                        MinimumY,
                        CurrentPosition.Z
                    )

                if Connection then
                    Connection:Disconnect()
                    Connection = nil
                end

                return false
            end
        end

        task.wait()

    end

    if Connection then
        Connection:Disconnect()
        Connection = nil
    end

    return ScriptRunning
end

local function RunAutoFarm()

    if AutoFarmThread then
        return
    end

    AutoFarmThread =
        task.spawn(function()

            while
                AutoFarm
                and ScriptRunning
            do

                ScanForPan()
                UpdateSandDisplay()

                if not CurrentPan then

                    task.wait(1)

                    continue
                end

                local Fill, Capacity =
                    GetSand()

                print(
                    "[AutoFarm] Sand:",
                    Fill,
                    "/",
                    Capacity
                )

                if Capacity <= 0 then

                    print(
                        "[AutoFarm] Capacity not detected."
                    )

                    task.wait(0.5)

                    continue
                end

                --==================================================
                -- SELECTED LOCATION
                --==================================================

                local Location =
                    LOCATIONS[SelectedLocation]

                if not Location then

                    warn(
                        "[AutoFarm] Selected location missing."
                    )

                    task.wait(1)

                    continue
                end

                local SandPosition =
                    Location.Sand

                local WaterPosition =
                    Location.Water

                print(
                    "[AutoFarm] Using location:",
                    Location.Name
                )

                --==================================================
                -- GO DIG
                --==================================================

                if Fill < Capacity then

                    print(
                        "[AutoFarm] Going to sand:",
                        Location.Name
                    )

                    if not TweenTo(
                        SandPosition
                    ) then

                        break
                    end

                    if not AutoFarm
                        or not ScriptRunning
                    then

                        break
                    end

                    while
                        AutoFarm
                        and ScriptRunning
                    do

                        ScanForPan()

                        local CurrentFill,
                            CurrentCapacity =
                            GetSand()

                        UpdateSandDisplay()

                        if not CurrentPan then
                            break
                        end

                        if CurrentCapacity <= 0 then

                            task.wait(0.5)

                            continue
                        end

                        if CurrentFill >=
                            CurrentCapacity
                        then

                            break
                        end

                        Collect()

                        task.wait(
                            AUTO_COLLECT_DELAY
                        )
                    end
                end

                if not AutoFarm
                    or not ScriptRunning
                then

                    break
                end

                ScanForPan()

                local FinalFill,
                    FinalCapacity =
                    GetSand()

                UpdateSandDisplay()

                if not CurrentPan then

                    task.wait(0.5)

                    continue
                end

                if FinalCapacity <= 0 then

                    task.wait(0.5)

                    continue
                end

                if FinalFill < FinalCapacity then

                    task.wait(0.2)

                    continue
                end

                --==================================================
                -- GO WATER
                --==================================================

                print(
                    "[AutoFarm] Pan full. Going to water:",
                    Location.Name
                )

                if not TweenTo(
                    WaterPosition
                ) then

                    break
                end

                if not AutoFarm
                    or not ScriptRunning
                then

                    break
                end

                task.wait(0.75)

                --==================================================
                -- START PANNING
                --==================================================

                local Started =
                    StartWaterPan()

                if not Started then

                    warn(
                        "[AutoFarm] Failed to start panning."
                    )

                    task.wait(1)

                    continue
                end

                -- SAME SHAKE MECHANIC

                ShakeWhilePanning()

                --==================================================
                -- WAIT UNTIL EMPTY
                --==================================================

                local EmptyTimeout =
                    os.clock() + 30

                while
                    AutoFarm
                    and ScriptRunning
                    and os.clock() < EmptyTimeout
                do

                    UpdateSandDisplay()

                    local CurrentFill,
                        CurrentCapacity =
                        GetSand()

                    if CurrentFill <= 0 then

                        print(
                            "[AutoFarm] Pan emptied."
                        )

                        break
                    end

                    local Character =
                        LocalPlayer.Character

                    if Character
                        and Character:GetAttribute(
                            "Panning"
                        ) ~= true
                    then

                        print(
                            "[AutoFarm] Panning stopped. Restarting."
                        )

                        task.wait(0.25)

                        local Restarted =
                            StartWaterPan()

                        if Restarted then
                            ShakeWhilePanning()
                        end
                    end

                    task.wait(0.1)

                end

                task.wait(0.25)

            end

            AutoFarmThread = nil

        end)
end
